# Deliberately no native audio dependency: all audio capture and playback happens in the
# browser, so this installs identically on Windows, Linux and macOS. grpcio (pulled in by
# nvidia-riva-client) ships wheels for win32, win_amd64, manylinux x86_64/aarch64 and macOS
# universal2, so no compiler is needed anywhere.
#
# Versions below are the ones actually verified working together on 2026-09-13. In particular
# websockets is pinned to 15.0.1 because nvidia-riva-client requires >=15.0.1,<16 — an earlier
# pin of 14.1 installed fine locally (pip had already upgraded it) but broke every fresh clone.
fastapi==0.115.6
uvicorn==0.34.0
websockets==15.0.1
httpx==0.28.1
nvidia-riva-client==2.27.0
python-dotenv==1.0.1

# Installs this project so `python -m coach` works. Last line: it must resolve after the pins
# above.
-e .
